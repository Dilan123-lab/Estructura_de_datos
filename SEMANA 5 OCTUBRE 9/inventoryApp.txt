package com.mycompany.inventoryapp;

import java.util.Scanner;

public class InventoryApp {

    public static void main(String[] args) {
        Inventory inventory = new Inventory();
        Scanner scanner = new Scanner(System.in);

        // Carga inicial de datos de prueba
        inventory.newProduct(1, "Manzana", 50, 2000.0, "Frutas");
        inventory.newProduct(2, "Banano", 70, 1300.0, "Frutas");
        inventory.newProduct(3, "Aguacate", 30, 6000.0, "Verduras");

        int option = 0;

        do {
            System.out.println("\n----- MENÚ DE INVENTARIO -----");
            System.out.println("1. Agregar producto nuevo");
            System.out.println("2. Aumentar existencia (+1)");
            System.out.println("3. Actualizar precio");
            System.out.println("4. Consultar producto por ID");
            System.out.println("5. Eliminar producto");
            System.out.println("6. Mostrar todos los productos");
            System.out.println("7. Salir");
            System.out.print("Seleccione una opción: ");
            
            option = scanner.nextInt();

            switch (option) {
                case 1:
                    System.out.print("Ingrese ID: ");
                    int id = scanner.nextInt();
                    scanner.nextLine();
                    System.out.print("Ingrese Nombre: ");
                    String name = scanner.nextLine();
                    System.out.print("Ingrese Existencias: ");
                    int existence = scanner.nextInt();
                    System.out.print("Ingrese Precio: ");
                    double price = scanner.nextDouble();
                    scanner.nextLine();
                    System.out.print("Ingrese Categoría: ");
                    String category = scanner.nextLine();

                    inventory.newProduct(id, name, existence, price, category);
                    break;

                case 2:
                    System.out.print("Ingrese el ID del producto para aumentar existencia: ");
                    int idAdd = scanner.nextInt();
                    inventory.addProduct(idAdd);
                    break;

                case 3:
                    System.out.print("Ingrese el ID del producto a actualizar precio: ");
                    int idUp = scanner.nextInt();
                    System.out.print("Ingrese el nuevo precio: ");
                    double newPrice = scanner.nextDouble();
                    inventory.updateProduct(idUp, newPrice);
                    break;

                case 4:
                    System.out.print("Ingrese el ID del producto a consultar: ");
                    int idSearch = scanner.nextInt();
                    inventory.searchProduct(idSearch);
                    break;

                case 5:
                    System.out.print("Ingrese el ID del producto a eliminar: ");
                    int idDel = scanner.nextInt();
                    inventory.deleteProduct(idDel);
                    break;

                case 6:
                    inventory.printProducts();
                    break;

                case 7:
                    System.out.println("Saliendo del programa...");
                    break;

                default:
                    System.out.println("Opción no válida.");
            }

        } while (option != 7);

        scanner.close();
    }
}
