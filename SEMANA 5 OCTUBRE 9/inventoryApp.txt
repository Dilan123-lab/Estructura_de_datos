package com.mycompany.inventoryapp;

import java.util.Objects;

public class Product {
    
    private int ID;
    private int Existence;
    private String name;
    private Double Price;
    private String category;

    // Constructor completo
    public Product(int ID, int Existence, String name, Double Price, String category) {
        this.ID = ID;
        this.Existence = Existence;
        this.name = name;
        this.Price = Price;
        this.category = category;
    }

    // Constructor vacío
    public Product() {
    }

    
    public Product(int ID) {
        this.ID = ID;
    }

    // Getters y Setters
    public int getID() {
        return ID;
    }

    public void setID(int ID) {
        this.ID = ID;
    }

    public int getExistence() {
        return Existence;
    }

    public void setExistence(int Existence) {
        this.Existence = Existence;
    }

    public String getName() {
        return name;
    }

    public void setName(String name) {
        this.name = name;
    }

    public Double getPrice() {
        return Price;
    }

    public void setPrice(Double Price) {
        this.Price = Price;
    }

    public String getCategory() {
        return category;
    }

    public void setCategory(String category) {
        this.category = category;
    }

    // Comparación por ID para que indexOf funcione
    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (o == null || getClass() != o.getClass()) return false;
        Product product = (Product) o;
        return ID == product.ID;
    }

    @Override
    public int hashCode() {
        return Objects.hash(ID);
    }

    @Override
    public String toString() {
        return "Product{" + "ID=" + ID + ", nombre=" + name + ", Existencias=" + Existence + ", Precio=" + Price + ", categoria=" + category + '}';
    }
}
